Nginx

server {
    server_name splitdine.express.noodev8.com;

    location / {
        proxy_pass http://localhost:3010; # YOUR PORT
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # Allow JSON content
        default_type application/json;
    }

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/splitdine.express.noodev8.com/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/splitdine.express.noodev8.com/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}
server {
    if ($host = splitdine.express.noodev8.com) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80;
    server_name splitdine.express.noodev8.com;
    return 404; # managed by Certbot


}
